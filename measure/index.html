const pushToCloud = async (throwOnError = false) => { const code = String(project.code || '').trim(); if (!db || !code) return; if (!navigator.onLine) { lastCloudError.value = formatSyncError(new Error('offline')); cloudSyncState.value = 'error'; if (throwOnError) throw new Error(lastCloudError.value); return; } cloudSyncState.value = 'syncing'; try { await trySilentAuth(); const projRef = db.collection('site_projects').doc(code); const now = Date.now(); const lastSync = project.lastSyncAt || 0; // 取得上次成功同步的時間 // 🌟 1. 篩選出「真正有動過」的分頁 // 如果是全新建立的工程，lastSync 為 0，就會全部上傳 const dirtyPagesRaw = pages.value.filter(pg => (pg.updatedAt || 0) > lastSync); // 清理不需要上傳的暫存資料 const dirtyPages = JSON.parse(JSON.stringify(dirtyPagesRaw)).map(pg => { delete pg.historyStack; delete pg.redoStack; (pg.pins || []).forEach(p => { delete p.historyStack; delete p.redoStack; }); return pg; }); const myPageIds = pages.value.map(p => p.id); const myDeletedIds = project.deletedPageIds || []; // 🌟 2. 更新主工程目錄 (使用 arrayUnion 避免洗掉別人新增的頁面) await withRetry(() => withTimeout(projRef.set({ code, name: project.name || '', date: project.date || '', // 用 arrayUnion 確保我們只會「加入」我們的 ID，不會覆蓋掉別人剛加的 ID pageIds: firebase.firestore.FieldValue.arrayUnion(...myPageIds), deletedPageIds: myDeletedIds.length ? firebase.firestore.FieldValue.arrayUnion(...myDeletedIds) : [], updatedAt: now }, { merge: true }))); // 🌟 3. 「只」上傳有改動的分頁到 pages 集合 for (const pg of dirtyPages) { const { photo: pagePhoto, pins, ...restPage } = pg; // 如果照片也有更新（通常是剛拍照或剛揀相），才上傳照片檔案 if (pagePhoto && !pg.isSketch) { await withRetry(() => withTimeout(projRef.collection('media').doc('page_' + pg.id).set({ photo: pagePhoto, updatedAt: pg.updatedAt || now }, { merge: true }))); } const pinsMeta = []; for (const pin of (pins || [])) { const { photo: pinPhoto, ...restPin } = pin; pinsMeta.push({ ...restPin, hasPhoto: !!pinPhoto }); // 釘選照片如果有改動，也獨立上傳 if (pinPhoto && !pin.isSketch && (pin.updatedAt || 0) > lastSync) { await withRetry(() => withTimeout(projRef.collection('media').doc(String(pin.id)).set({ photo: pinPhoto, updatedAt: pin.updatedAt || now }, { merge: true }))); } } // 儲存該頁面的線條、印章等文字數據 await withRetry(() => withTimeout(projRef.collection('pages').doc(pg.id).set({ ...restPage, hasPhoto: !!pagePhoto, pins: pinsMeta }, { merge: true }))); } lastCloudError.value = ''; cloudSyncState.value = 'ready'; // 🌟 4. 上傳成功後，更新本地的最後同步時間，這樣下次沒改動就不會再傳 project.lastSyncAt = now; const d = await localforage.getItem('proj_' + code); if (d) { d.lastSyncAt = now; await localforage.setItem('proj_' + code, d); } } catch (e) { lastCloudError.value = formatSyncError(e); cloudSyncState.value = 'error'; if (throwOnError) throw new Error(lastCloudError.value); } };
